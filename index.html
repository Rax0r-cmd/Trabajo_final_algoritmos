// Arquitectura POO (Herencia + Interfaz) 
public interface IReservable {
    bool PuedeReservarse(DateTime inicio, DateTime fin);
    decimal CalcularCosto(TimeSpan duracion);
}

public abstract class Recurso : IReservable {
    public int Id { get; set; }
    public string Nombre { get; set; }
    public string Ubicacion { get; set; }
    public decimal CostoPorHora { get; set; }
    public abstract decimal CalcularCosto(TimeSpan duracion);
    public virtual bool PuedeReservarse(DateTime inicio, DateTime fin) => true;
}

public class Sala : Recurso {
    public int CapacidadPersonas { get; set; }
    public bool TieneProyector { get; set; }
    public override decimal CalcularCosto(TimeSpan duracion) => 
        (decimal)duracion.TotalHours * CostoPorHora * 1.10m;
}
